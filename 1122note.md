针对 **Python + FastAPI + PostgreSQL** 技术栈，解决“各轮次文件要求不同（如初赛 1 个 PPT + 1 个 PDF）”的最佳实践是：**在 config 表中用 JSONB 定义“文件槽位（File Slots）”规则，在 submissions 表中用 JSONB 字典按 key 存储文件**。
这样无论后续复赛、决赛改成什么格式组合（例如复赛要视频+源码压缩包，决赛要终版报告），**数据库表结构和 API 接口零改动**，只需在后台改一下 JSON 配置即可。
以下是可直接落地的完整架构代码。
## 1. PostgreSQL 表结构与 SQLAlchemy 模型 (models.py)
利用 PostgreSQL 原生的 JSONB 类型存储动态文件规则、用户上传的文件映射以及各轮次独立的 AI 反馈。
```python
import enum
from datetime import datetime
from sqlalchemy import Column, Integer, String, Text, DateTime, ForeignKey, UniqueConstraint, func
from sqlalchemy.dialects.postgresql import JSONB
from sqlalchemy.orm import declarative_base, relationship

Base = declarative_base()

class GlobalStage(str, enum.Enum):
    ALL_CLOSED = "ALL_CLOSED"       # 全程关闭
    PRELIM_OPEN = "PRELIM_OPEN"     # 开放初赛
    PRELIM_ENDED = "PRELIM_ENDED"   # 初赛结束
    SEMI_OPEN = "SEMI_OPEN"         # 开放复赛
    SEMI_ENDED = "SEMI_ENDED"       # 复赛结束
    FINAL_OPEN = "FINAL_OPEN"       # 开放决赛
    FINAL_ENDED = "FINAL_ENDED"     # 决赛结束

class SubmissionStatus(str, enum.Enum):
    REGISTERED = "REGISTERED"           # 报名（初赛资格）
    ADVANCED_SEMI = "ADVANCED_SEMI"     # 晋级复赛
    ADVANCED_FINAL = "ADVANCED_FINAL"   # 晋级决赛

class RoundName(str, enum.Enum):
    PRELIM = "PRELIM"
    SEMI = "SEMI"
    FINAL = "FINAL"

# 1. 全局赛程与各轮次配置表（单行配置表，id固定为1）
class CompetitionConfig(Base):
    __tablename__ = "competition_configs"

    id = Column(Integer, primary_key=True, default=1)
    current_stage = Column(String(32), nullable=False, default=GlobalStage.ALL_CLOSED.value)
    
    # 存储每一轮的 {"file_requirements": [...], "ai_prompt": "..."}
    prelim_config = Column(JSONB, nullable=False, default=dict)
    semi_config = Column(JSONB, nullable=False, default=dict)
    final_config = Column(JSONB, nullable=False, default=dict)
    
    updated_at = Column(DateTime(timezone=True), server_default=func.now(), onupdate=func.now())

# 2. 投稿主表（保留原有CRUD核心字段）
class Submission(Base):
    __tablename__ = "submissions"

    id = Column(Integer, primary_key=True, index=True)
    user_id = Column(Integer, nullable=False, index=True)
    title = Column(String(255), nullable=False)
    description = Column(Text, nullable=True)
    
    # 参赛状态：默认报名
    status = Column(String(32), nullable=False, default=SubmissionStatus.REGISTERED.value)
    
    # 所有轮次上传的文件汇总字典，格式:
    # {
    #   "prelim_ppt": {"url": "https://...", "filename": "路演.pptx"},
    #   "prelim_pdf": {"url": "https://...", "filename": "计划书.pdf"}
    # }
    files = Column(JSONB, nullable=False, default=dict)
    
    created_at = Column(DateTime(timezone=True), server_default=func.now())
    updated_at = Column(DateTime(timezone=True), server_default=func.now(), onupdate=func.now())
    deleted_at = Column(DateTime(timezone=True), nullable=True)  # 软删除标记

    stage_records = relationship("SubmissionStageRecord", back_populates="submission", cascade="all, delete-orphan")

# 3. 轮次隔离记录表（专门隔离每一轮的 ai_message）
class SubmissionStageRecord(Base):
    __tablename__ = "submission_stage_records"

    id = Column(Integer, primary_key=True)
    submission_id = Column(Integer, ForeignKey("submissions.id", ondelete="CASCADE"), nullable=False)
    stage = Column(String(16), nullable=False)  # PRELIM / SEMI / FINAL
    
    ai_message = Column(Text, nullable=True)    # 仅存本轮的 AI 反馈
    updated_at = Column(DateTime(timezone=True), server_default=func.now(), onupdate=func.now())

    submission = relationship("Submission", back_populates="stage_records")

    __table_args__ = (
        UniqueConstraint("submission_id", "stage", name="uq_submission_stage"),
    )

```

### 动态文件校验逻辑 (validator.py)
既能校验当前轮次的必填项和后缀名，又能在用户在复赛修改初赛旧文件时，确保旧文件依然符合初赛的格式规范：
```python
import os
from typing import Dict, Any
from fastapi import HTTPException
from models import CompetitionConfig, GlobalStage, RoundName, SubmissionStatus

STAGE_RULES = {
    GlobalStage.PRELIM_OPEN.value:  {"round": RoundName.PRELIM.value, "writable": True,  "req_status": SubmissionStatus.REGISTERED.value},
    GlobalStage.PRELIM_ENDED.value: {"round": RoundName.PRELIM.value, "writable": False, "req_status": None},
    GlobalStage.SEMI_OPEN.value:    {"round": RoundName.SEMI.value,   "writable": True,  "req_status": SubmissionStatus.ADVANCED_SEMI.value},
    GlobalStage.SEMI_ENDED.value:   {"round": RoundName.SEMI.value,   "writable": False, "req_status": None},
    GlobalStage.FINAL_OPEN.value:   {"round": RoundName.FINAL.value,  "writable": True,  "req_status": SubmissionStatus.ADVANCED_FINAL.value},
    GlobalStage.FINAL_ENDED.value:  {"round": RoundName.FINAL.value,  "writable": False, "req_status": None},
    GlobalStage.ALL_CLOSED.value:   {"round": None,                   "writable": False, "req_status": None},
}

def get_round_config(config: CompetitionConfig, round_name: str) -> dict:
    mapping = {
        RoundName.PRELIM.value: config.prelim_config,
        RoundName.SEMI.value: config.semi_config,
        RoundName.FINAL.value: config.final_config,
    }
    return mapping.get(round_name, {})

def validate_files_for_stage(config: CompetitionConfig, current_round: str, merged_files: Dict[str, Any]):
    """
    1. 校验当前轮次所有 required=True 的文件槽位是否齐全、格式是否正确
    2. 校验用户顺带修改的往期文件是否符合其原本的格式要求
    """
    current_cfg = get_round_config(config, current_round)
    current_reqs = current_cfg.get("file_requirements", [])

    # 汇总所有轮次的文件规则映射表（用于校验修改历史文件时的后缀名）
    all_rules_map = {}
    for r in [RoundName.PRELIM.value, RoundName.SEMI.value, RoundName.FINAL.value]:
        for item in get_round_config(config, r).get("file_requirements", []):
            all_rules_map[item["key"]] = item

    # 1. 检查当前轮次的必填槽位
    for req in current_reqs:
        key = req["key"]
        if req.get("required") and key not in merged_files:
            raise HTTPException(status_code=400, detail=f"缺少当前阶段必填文件：{req['label']} ({key})")

    # 2. 检查所有已提交/修改的文件后缀名是否合法
    for file_key, file_info in merged_files.items():
        rule = all_rules_map.get(file_key)
        if not rule:
            raise HTTPException(status_code=400, detail=f"未知的文件字段标识：{file_key}")
        
        filename = file_info.get("filename", "")
        ext = os.path.splitext(filename)[1].lower()
        allowed_exts = [e.lower() for e in rule.get("allowed_extensions", [])]
        
        if allowed_exts and ext not in allowed_exts:
            raise HTTPException(
                status_code=400,
                detail=f"文件 [{rule['label']}] 格式错误，仅支持 {', '.join(allowed_exts)}，当前为 {ext}"
            )

```
## 3. 用户端 5 个核心 API 实现 (router.py)
完全保留你原有的路由路径，通过统一装配函数 assemble_submission_response 实现“修改历史文件”与“仅返回本轮 ai_message”。
```python
from datetime import datetime, timezone
from typing import Dict, Optional
from fastapi import APIRouter, Depends, HTTPException
from pydantic import BaseModel
from sqlalchemy.orm import Session

from models import CompetitionConfig, Submission, SubmissionStageRecord, GlobalStage, SubmissionStatus
from validator import STAGE_RULES, get_round_config, validate_files_for_stage

router = APIRouter()

# --- Pydantic Schemas ---
class FileItem(BaseModel):
    url: str
    filename: str

class SubmissionCreate(BaseModel):
    title: str
    description: Optional[str] = None
    files: Dict[str, FileItem]  # 例如: {"prelim_ppt": {"url": "...", "filename": "a.pptx"}, "prelim_pdf": {...}}

class SubmissionUpdate(BaseModel):
    title: Optional[str] = None
    description: Optional[str] = None
    files: Optional[Dict[str, FileItem]] = None  # 支持只传改动的文件，也支持全量传

# --- 模拟依赖项（替换为你的实际实现） ---
def get_db(): ...
def get_current_user(): ...
async def generate_ai_feedback(prompt: str, files: dict) -> str:
    # 调用大模型 API，传入本轮 prompt 和文件列表，返回评语
    return "AI 评估反馈内容..."

# --- 核心：响应装配器（隔离各轮 ai_message） ---
def assemble_submission_response(sub: Submission, config: CompetitionConfig, db: Session) -> dict:
    stage_ctx = STAGE_RULES.get(config.current_stage, {})
    current_round = stage_ctx.get("round")
    
    # 1. 只查询当前轮次的 ai_message（全程关闭或无记录时返回 None）
    current_ai_msg = None
    if current_round:
        record = db.query(SubmissionStageRecord).filter_by(
            submission_id=sub.id, stage=current_round
        ).first()
        if record:
            current_ai_msg = record.ai_message

    # 2. 计算当前单条投稿是否可编辑/可删除
    can_edit = (
        stage_ctx.get("writable", False)
        and sub.status == stage_ctx.get("req_status")
    )

    # 3. 获取当前轮次的文件配置，供前端动态渲染上传框
    round_cfg = get_round_config(config, current_round) if current_round else {}

    return {
        "id": sub.id,
        "title": sub.title,
        "description": sub.description,
        "status": sub.status,
        "global_stage": config.current_stage,
        "can_edit": can_edit,
        "can_delete": can_edit,
        "files": sub.files,  # 返回全量文件，方便用户在复赛/决赛修改之前轮次的文件
        "current_file_requirements": round_cfg.get("file_requirements", []),
        "ai_message": current_ai_msg,  # 严格只返回本轮 AI 消息
        "updated_at": sub.updated_at,
    }

# ==========================================
# 1. 获取用户自己的全部投稿
# ==========================================
@router.get("/submissions")
def list_my_submissions(db: Session = Depends(get_db), user=Depends(get_current_user)):
    config = db.query(CompetitionConfig).first()
    subs = db.query(Submission).filter(
        Submission.user_id == user.id,
        Submission.deleted_at.is_(None)
    ).order_by(Submission.id.desc()).all()

    round_cfg = get_round_config(config, STAGE_RULES[config.current_stage]["round"])
    
    return {
        "global_stage": config.current_stage,
        "can_create": config.current_stage == GlobalStage.PRELIM_OPEN.value,
        "current_file_requirements": round_cfg.get("file_requirements", []),
        "items": [assemble_submission_response(s, config, db) for s in subs]
    }

# ==========================================
# 2. 获取单一投稿详情
# ==========================================
@router.get("/submission/{submission_id}")
def get_submission_detail(submission_id: int, db: Session = Depends(get_db), user=Depends(get_current_user)):
    config = db.query(CompetitionConfig).first()
    sub = db.query(Submission).filter(
        Submission.id == submission_id,
        Submission.deleted_at.is_(None)
    ).first()
    
    if not sub:
        raise HTTPException(status_code=404, detail="投稿不存在")
    if not user.is_admin and sub.user_id != user.id:
        raise HTTPException(status_code=403, detail="无权查看该投稿")

    return assemble_submission_response(sub, config, db)

# ==========================================
# 3. 新建投稿（仅初赛开放期可用）
# ==========================================
@router.post("/submissions")
async def create_submission(payload: SubmissionCreate, db: Session = Depends(get_db), user=Depends(get_current_user)):
    config = db.query(CompetitionConfig).first()
    if config.current_stage != GlobalStage.PRELIM_OPEN.value:
        raise HTTPException(status_code=403, detail="当前不在初赛报名提交时间")

    files_dict = {k: v.model_dump() for k, v in payload.files.items()}
    validate_files_for_stage(config, "PRELIM", files_dict)

    sub = Submission(
        user_id=user.id,
        title=payload.title,
        description=payload.description,
        status=SubmissionStatus.REGISTERED.value,
        files=files_dict
    )
    db.add(sub)
    db.flush()  # 获取 sub.id

    # 触发初赛 AI 评估并写入 PRELIM 记录
    prelim_cfg = get_round_config(config, "PRELIM")
    ai_msg = await generate_ai_feedback(prelim_cfg.get("ai_prompt", ""), files_dict)
    
    db.add(SubmissionStageRecord(submission_id=sub.id, stage="PRELIM", ai_message=ai_msg))
    db.commit()
    db.refresh(sub)

    return assemble_submission_response(sub, config, db)

# ==========================================
# 4. 修改投稿（支持修改历史文件，仅触发本轮AI）
# ==========================================
@router.put("/submission/{submission_id}")
async def update_submission(
    submission_id: int,
    payload: SubmissionUpdate,
    db: Session = Depends(get_db),
    user=Depends(get_current_user)
):
    config = db.query(CompetitionConfig).first()
    stage_ctx = STAGE_RULES.get(config.current_stage, {})

    if not stage_ctx.get("writable"):
        raise HTTPException(status_code=403, detail="当前赛程已锁定或关闭，无法修改")

    sub = db.query(Submission).filter(
        Submission.id == submission_id,
        Submission.user_id == user.id,
        Submission.deleted_at.is_(None)
    ).first()
    if not sub:
        raise HTTPException(status_code=404, detail="投稿不存在")

    # 校验晋级资格：例如复赛开放期，只有 status == ADVANCED_SEMI 的稿件才能改
    if sub.status != stage_ctx.get("req_status"):
        raise HTTPException(status_code=403, detail=f"当前投稿状态({sub.status})无权在 {config.current_stage} 阶段修改")

    current_round = stage_ctx["round"]

    # 合并文件：保留未修改的历史文件，覆盖新传的文件（实现“每次开放可以修改之前的”）
    incoming_files = {k: v.model_dump() for k, v in (payload.files or {}).items()}
    merged_files = {**(sub.files or {}), **incoming_files}

    # 校验合并后的文件是否满足当前轮次的格式与必填要求
    validate_files_for_stage(config, current_round, merged_files)

    if payload.title is not None:
        sub.title = payload.title
    if payload.description is not None:
        sub.description = payload.description
    sub.files = merged_files

    # 仅使用当前轮次的 prompt 评估，并只更新当前轮次的 ai_message
    round_cfg = get_round_config(config, current_round)
    ai_msg = await generate_ai_feedback(round_cfg.get("ai_prompt", ""), merged_files)

    record = db.query(SubmissionStageRecord).filter_by(
        submission_id=sub.id, stage=current_round
    ).first()
    if record:
        record.ai_message = ai_msg
    else:
        db.add(SubmissionStageRecord(submission_id=sub.id, stage=current_round, ai_message=ai_msg))

    db.commit()
    db.refresh(sub)

    return assemble_submission_response(sub, config, db)

# ==========================================
# 5. 软删除投稿
# ==========================================
@router.delete("/submission/{submission_id}")
def delete_submission(submission_id: int, db: Session = Depends(get_db), user=Depends(get_current_user)):
    config = db.query(CompetitionConfig).first()
    stage_ctx = STAGE_RULES.get(config.current_stage, {})

    if not stage_ctx.get("writable"):
        raise HTTPException(status_code=403, detail="当前赛程未开放，禁止删除投稿")

    sub = db.query(Submission).filter(
        Submission.id == submission_id,
        Submission.user_id == user.id,
        Submission.deleted_at.is_(None)
    ).first()
    if not sub:
        raise HTTPException(status_code=404, detail="投稿不存在")

    if sub.status != stage_ctx.get("req_status"):
        raise HTTPException(status_code=403, detail="当前投稿状态无权在此阶段删除")

    sub.deleted_at = datetime.now(timezone.utc)
    db.commit()

    return {"message": "删除成功", "id": submission_id}

```
## 4. 关键设计细节说明
 1. **SQLAlchemy 修改 JSONB 字典的注意事项**：在 PUT 接口中，代码写的是 sub.files = merged_files（直接赋值一个新字典对象），这样 SQLAlchemy 能自动检测到字段变更并生成 UPDATE SQL。切勿直接调用 sub.files.update(...) 原地修改，否则 PostgreSQL JSONB 可能不会触发脏检查。
 2. **文件上传接口配合方式**：前端先调用你们现有的文件上传接口（如 POST /upload）把 PPT 和 PDF 传到对象存储或本地目录，拿到返回的 url 和原始 filename，再组装成 JSON 调 POST /submissions 或 PUT /submission/{id}。
 3. **管理员切换赛程**：你的 Admin 端只需要提供两个操作——修改 competition_configs.current_stage（切换赛程阶段），以及批量修改 submissions.status（晋级名单操作：REGISTERED \rightarrow ADVANCED_SEMI \rightarrow ADVANCED_FINAL），用户端的 5 个接口就会自动切换权限和返回对应的 ai_message。
